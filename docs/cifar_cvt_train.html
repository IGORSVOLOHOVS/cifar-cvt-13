 21# --- Monadic Error Handling ---
 22T = TypeVar("T")
 23E = TypeVar("E")
 24
 25@dataclass(frozen=True)
 26class Result(Generic[T, E]):
 27    """Generic Result type for monadic error handling."""
 28    value: T | None = None
 29    error: E | None = None
 30
 31    @classmethod
 32    def ok(cls, value: T) -> "Result[T, E]":
 33        return cls(value=value)
 34
 35    @classmethod
 36    def fail(cls, error: E) -> "Result[T, E]":
 37        return cls(error=error)
 38
 39    def is_ok(self) -> bool:
 40        return self.error is None
 41
 42    def unwrap(self) -> T:
 43        if self.error is not None:
 44            raise ValueError(f"Called unwrap on Failure: {self.error}")
 45        return cast(T, self.value)

 68# --- Core Logic ---
 69class CvtTrainer:
 70    """Functional core for CvT model training and evaluation."""
 71
 72    def __init__(self, config: ModelConfig) -> None:
 73        self.config = config
 74        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
 75        self.image_processor = AutoImageProcessor.from_pretrained(config.model_name) # type: ignore
 76        logger.info(f"Initialized CvtTrainer on [bold cyan]{self.device}[/bold cyan]")
 77
 78    def transform(self, image: Any) -> torch.Tensor:
 79        """Apply CvT-specific transformations."""
 80        inputs = self.image_processor(images=image, return_tensors="pt")
 81        return inputs["pixel_values"].squeeze(0) # type: ignore
 82
 83    def prepare_data(self) -> Result[tuple[DataLoader[Any], DataLoader[Any], DataLoader[Any], list[str]], str]:
 84        """Load and split CIFAR-10 data."""
 85        try:
 86            logger.info("Loading CIFAR-10 dataset...")
 87            full_train = datasets.CIFAR10(
 88                root=str(self.config.data_dir), train=True, download=True, transform=self.transform
 89            )
 90            test_dataset = datasets.CIFAR10(
 91                root=str(self.config.data_dir), train=False, download=True, transform=self.transform
 92            )
 93
 94            train_size = len(full_train) - self.config.val_size
 95            train_dataset, val_dataset = random_split(full_train, [train_size, self.config.val_size])
 96
 97            loaders: tuple[DataLoader[Any], DataLoader[Any], DataLoader[Any], list[str]] = (
 98                DataLoader(train_dataset, batch_size=self.config.batch_size, shuffle=True),
 99                DataLoader(val_dataset, batch_size=self.config.batch_size, shuffle=False),
100                DataLoader(test_dataset, batch_size=self.config.batch_size, shuffle=False),
101                list(full_train.classes)
102            )
103            logger.info(f"Data ready: {train_size} train, {self.config.val_size} val, {len(test_dataset)} test.")
104            return Result[tuple[DataLoader[Any], DataLoader[Any], DataLoader[Any], list[str]], str].ok(loaders)
105        except Exception as e:
106            return Result[tuple[DataLoader[Any], DataLoader[Any], DataLoader[Any], list[str]], str].fail(f"Failed to load data: {e}")
107
108    def initialize_model(self, num_classes: int) -> Result[nn.Module, str]:
109        """Load and prepare CvtForImageClassification."""
110        try:
111            logger.info(f"Loading model {self.config.model_name}...")
112            model = CvtForImageClassification.from_pretrained(self.config.model_name)
113            model.classifier = nn.Linear(model.classifier.in_features, num_classes) # type: ignore
114            model.to(self.device) # type: ignore
115            return Result[nn.Module, str].ok(model)
116        except Exception as e:
117            return Result[nn.Module, str].fail(f"Failed to initialize model: {e}")
118
119    def train_epoch(self, model: nn.Module, loader: DataLoader[Any]) -> float:
120        """Train model for one epoch."""
121        model.train()
122        optimizer = torch.optim.AdamW(model.parameters(), lr=self.config.lr)
123        criterion = nn.CrossEntropyLoss()
124        total_loss = 0.0
125
126        with Progress(
127            SpinnerColumn(),
128            TextColumn("[progress.description]{task.description}"),
129            BarColumn(),
130            TextColumn("[progress.percentage]{task.percentage:>3.0f}%"),
131            TimeElapsedColumn(),
132            console=console
133        ) as progress:
134            task = progress.add_task("Training...", total=len(loader))
135            
136            for images, labels in loader:
137                images, labels = images.to(self.device), labels.to(self.device)
138                
139                optimizer.zero_grad()
140                outputs = model(images)
141                loss = criterion(outputs.logits, labels)
142                loss.backward()
143                optimizer.step()
144                
145                total_loss += loss.item()
146                progress.update(task, advance=1, description=f"Loss: {loss.item():.4f}")
147
148        return total_loss / len(loader)
149
150    def evaluate(self, model: nn.Module, loader: DataLoader[Any], classes: list[str]) -> float:
151        """Evaluate model and print metrics."""
152        model.eval()
153        all_preds: list[int] = []
154        all_true: list[int] = []
155
156        with torch.no_grad():
157            for images, labels in loader:
158                images = images.to(self.device)
159                outputs = model(images)
160                _, predicted = torch.max(outputs.logits, 1)
161                all_preds.extend(predicted.cpu().numpy().tolist())
162                all_true.extend(labels.numpy().tolist())
163
164        accuracy = accuracy_score(all_true, all_preds)
165        logger.info("\n[bold]Classification Report:[/bold]")
166        console.print(classification_report(all_true, all_preds, target_names=classes))
167        return float(accuracy)

169def main() -> None:
170    """Imperative shell for the training pipeline."""
171    config = ModelConfig()
172    trainer = CvtTrainer(config)
173
174    # Data Path
175    data_res = trainer.prepare_data()
176    if not data_res.is_ok():
177        logger.error(data_res.error)
178        return
179    train_loader, val_loader, test_loader, classes = data_res.unwrap()
180
181    # Model Path
182    model_res = trainer.initialize_model(len(classes))
183    if not model_res.is_ok():
184        logger.error(model_res.error)
185        return
186    model = model_res.unwrap()
187
188    # Training Path
189    for epoch in range(config.num_epochs):
190        logger.info(f"Starting Epoch {epoch+1}/{config.num_epochs}")
191        avg_loss = trainer.train_epoch(model, train_loader)
192        logger.info(f"Epoch {epoch+1} complete. Average Loss: [bold red]{avg_loss:.4f}[/bold red]")
193
194    # Eval Path
195    final_acc = trainer.evaluate(model, test_loader, classes)
196    logger.info(f"Final Accuracy: [bold green]{final_acc:.4f}[/bold green]")
197
198    if final_acc > 0.65:
199        logger.info("[bold green]SUCCESS: Accuracy is above 65%.[/bold green]")
200    else:
201        logger.warning("Accuracy is below target (65%). Consider more epochs.")
202
203if __name__ == "__main__":
204    main()
